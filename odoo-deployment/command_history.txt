# Commandes exécutées pour le TP Docker Compose Odoo 18

## Étape 1 : Création docker-compose.yml initial
cat <<'YML' > docker-compose.yml
version: '3'
services:
  odoo-postgres:
    hostname: odoo-postgres
    image: postgres:15
    environment:
      - POSTGRES_USER=odoo
      - POSTGRES_PASSWORD=odoo123
      - POSTGRES_DB=postgres
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - odoo-network
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U odoo"]
      interval: 10s
      timeout: 5s
      retries: 5

  odoo-server:
    hostname: odoo-server
    image: odoo:18
    environment:
      - HOST=odoo-postgres
      - PORT=5432
      - USER=odoo
      - PASSWORD=odoo123
    volumes:
      - odoo_data:/var/lib/odoo
      - odoo_config:/etc/odoo
    networks:
      - odoo-network
    ports:
      - "8069:8069"
      - "8072:8072"

  odoo-pgadmin:
    hostname: odoo-pgadmin
    image: dpage/pgadmin4
    environment:
      - PGADMIN_DEFAULT_EMAIL=admin@example.com
      - PGADMIN_DEFAULT_PASSWORD=SuperSecret123!
      - PGADMIN_CONFIG_SERVER_MODE=False
      - PGADMIN_CONFIG_ENHANCED_COOKIES_ENABLED=False
      - PGADMIN_CONFIG_WTF_CSRF_ENABLED=False
    volumes:
      - pgadmin_data:/var/lib/pgadmin
    networks:
      - odoo-network
    ports:
      - "5050:80"

volumes:
  postgres_data:
    driver: local
    name: odoo_postgres_data
  odoo_data:
    driver: local
    name: odoo_app_data
  odoo_config:
    driver: local
    name: odoo_config_data
  pgadmin_data:
    driver: local
    name: pgadmin_config_data

networks:
  odoo-network:
    driver: bridge
    name: odoo-network
YML

## Étape 2 : Démarrage initial
docker compose up -d

## Vérifications :
docker compose ps
curl -s -o /dev/null -w "HTTP Status Odoo: %{http_code}" http://localhost:8069
curl -s -o /dev/null -w "HTTP Status PgAdmin: %{http_code}" http://localhost:5050/login
docker exec odoo-deployment-odoo-postgres-1 pg_isready -U odoo

## Étape 3 : Ajout restart: always et depends_on

# Modification du fichier :
# - Ajout de "restart: always" pour tous les services
# - Ajout de "depends_on: [odoo-postgres]" pour odoo-server et odoo-pgadmin
# - Changement port PostgreSQL 5432:5432 -> 5433:5432 (conflit port système)
